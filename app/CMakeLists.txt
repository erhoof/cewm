cmake_minimum_required(VERSION 3.13)
project(cewm C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(ANDROID)
  set(CEWM_DYNARMIC OFF CACHE BOOL "" FORCE)  # interpreter only, like the Aurora build
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-include${CMAKE_CURRENT_SOURCE_DIR}/android/shim/android_compat.h>)
endif()

# Upstream libCerfEngine needs CMake 3.25 (Aurora SDK has 3.19), so engine/ builds its sources itself; hw/ is CERF's board/SoC code.
# Baseline ARMv8.0 + CRC32 so one binary runs on every device (the MT8788 tablet is A53/A73: no LSE atomics, no fp16); tuned for the big cores.
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
  add_compile_options(-march=armv8-a+crc -mtune=cortex-a76)
endif()
if(ANDROID)
  # SDL2 comes from source (SDLActivity needs libSDL2.so beside libmain.so); the APK project lives in android/.
  set(SDL_STATIC OFF CACHE BOOL "" FORCE)
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../android-ports/SDL SDL2 EXCLUDE_FROM_ALL)
  target_compile_options(SDL2 PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/android/shim/sdl_looper_compat.h)
  set(SDL2_LIBRARIES SDL2)
  set(SDL2_INCLUDE_DIRS ${CMAKE_CURRENT_SOURCE_DIR}/../../android-ports/SDL/include)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED sdl2)
endif()
add_subdirectory(engine)
add_subdirectory(hw)


add_library(imgui STATIC
  third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
  third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp
  third_party/imgui/backends/imgui_impl_sdl2.cpp
  third_party/imgui/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui PUBLIC third_party/imgui third_party/imgui/backends ${SDL2_INCLUDE_DIRS})
target_compile_definitions(imgui PUBLIC IMGUI_IMPL_OPENGL_ES2)

if(ANDROID)
  add_library(cewm SHARED src/main.cpp src/launcher/theme.cpp host/haptic.cpp host/device_stats.cpp)  # SDLActivity loads libmain.so
  set_target_properties(cewm PROPERTIES OUTPUT_NAME main)
else()
  add_executable(cewm src/main.cpp src/launcher/theme.cpp host/haptic.cpp)
  pkg_check_modules(DBUS dbus-1)
endif()
if(ANDROID)
elseif(DBUS_FOUND)
  target_sources(cewm PRIVATE host/device_stats.cpp)
  target_compile_definitions(cewm PRIVATE CEWM_HAVE_DBUS)
  target_include_directories(cewm PRIVATE ${DBUS_INCLUDE_DIRS})
  target_link_directories(cewm PRIVATE ${DBUS_LIBRARY_DIRS})
  target_link_libraries(cewm PRIVATE ${DBUS_LIBRARIES})
else()
  target_sources(cewm PRIVATE host/device_stats.cpp)
endif()
# wl_surface_set_buffer_transform for the screen rotation hint (a no-op unless SDL runs on Wayland)
if(NOT ANDROID)
  pkg_check_modules(WAYLAND wayland-client)
endif()
if(WAYLAND_FOUND)
  target_compile_definitions(cewm PRIVATE CEWM_WAYLAND)
  target_include_directories(cewm PRIVATE ${WAYLAND_INCLUDE_DIRS})
  target_link_directories(cewm PRIVATE ${WAYLAND_LIBRARY_DIRS})
  target_link_libraries(cewm PRIVATE ${WAYLAND_LIBRARIES})
endif()
target_include_directories(cewm PRIVATE src/launcher)
target_link_directories(cewm PRIVATE ${SDL2_LIBRARY_DIRS})
target_include_directories(cewm PRIVATE host)
# Whole-archive: CERF services register themselves from static constructors.
target_link_libraries(cewm PRIVATE -Wl,--whole-archive cerfhw cerfengine -Wl,--no-whole-archive imgui ${SDL2_LIBRARIES} GLESv2 EGL dl $<IF:$<BOOL:${ANDROID}>,log;android,pthread>)

option(CEWM_BUILD_TESTS "Build headless emulator and CPU checks" OFF)
if(CEWM_BUILD_TESTS)
  enable_testing()
  add_executable(ril_modem_test test/ril_modem_test.cpp)
  target_include_directories(ril_modem_test PRIVATE host)
  add_test(NAME ril_modem COMMAND ril_modem_test)
  add_executable(cewm-probe host/emu_test_main.cpp host/haptic.cpp)
  target_include_directories(cewm-probe PRIVATE host)
  target_link_libraries(cewm-probe PRIVATE -Wl,--whole-archive cerfhw cerfengine -Wl,--no-whole-archive ${SDL2_LIBRARIES} $<IF:$<BOOL:${ANDROID}>,log;android,pthread>)
  if(CEWM_UNICORN_DIR)
    add_executable(unicorn_test test/unicorn_test.cpp)
    target_include_directories(unicorn_test PRIVATE "${CEWM_UNICORN_DIR}/include")
    target_link_libraries(unicorn_test PRIVATE "${CEWM_UNICORN_LIBRARY}" m)
    add_test(NAME unicorn_cpu_mmu COMMAND unicorn_test)
  endif()
endif()
